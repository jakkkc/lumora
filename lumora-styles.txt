export const metadata: Metadata = { title: "Accounts" };

function Breakdown({ title, rows }: { title: string; rows: Group[] }) {
  const max = Math.max(...rows.map((r) => r.total), 1);
  return (
    <section aria-label={title}>
      <h2 className="mb-3 text-xl">{title}</h2>
      {rows.every((r) => r.count === 0) ? (
        <p className="text-sm text-muted">No sales in this period.</p>
      ) : (
        <ul className="divide-y divide-line border-y border-line">
          {rows.map((r) => (
            <li key={r.name} className="py-3 text-sm">
              <div className="flex justify-between gap-4">
                <span>
                  {r.name}
                  <span className="text-muted">, {r.count} {r.count === 1 ? "booking" : "bookings"}</span>
                </span>
                <span>{money(r.total, "KES")}</span>
              </div>
              <div className="mt-2 h-1 bg-line">
                <div className="h-1 bg-accent" style={{ width: `${(r.total / max) * 100}%` }} />
              </div>
            </li>
          ))}
        </ul>
      )}
    </section>
  );
}

export default async function AccountsPage({
  searchParams,
}: {
  searchParams: Promise<{ range?: string; from?: string; to?: string; view?: string }>;
}) {
  const sp = await searchParams;
  const range = resolveRange(sp);
  const view = sp.view === "transactions" ? "transactions" : "sales";
  const a = await loadAccounts(range);
  const base = rangeQuery(range);

  const cards = [
    { label: "Sales", value: a.salesTotal, note: `${a.sales.length} completed ${a.sales.length === 1 ? "booking" : "bookings"}` },
    { label: "Money received", value: a.received, note: `${a.receivedCount} ${a.receivedCount === 1 ? "payment" : "payments"}` },
    { label: "Refunds", value: a.refunded, note: "Given back to guests" },
    { label: "Net received", value: a.net, note: "Received minus refunds" },
  ];

  const tab = (on: boolean) =>
    `nav-link tracking-wider ${on ? "" : "text-muted"}`;

  return (
    <div className="space-y-10">
      <div>
        <h1 className="text-3xl">accounts</h1>
        <p className="mt-2 text-sm text-muted">{rangeLabel(range)}</p>
      </div>

      <DateFilter range={range} view={view} />

            <p className="mt-2 text-xl sm:text-2xl">{money(c.value, "KES")}</p>
            <p className="mt-1 text-xs text-muted">{c.note}</p>

function baseEstimate(r: Req): { amount: number; note: string; currency: string } | null {
  const span = daysBetween(r.starts_on, r.ends_on);
  if (r.kind === "stay" && r.property) {
    const n = Math.max(span, 1);
    return { amount: n * r.property.price_per_night, note: `${n} night${n === 1 ? "" : "s"}`, currency: r.property.currency };
  }
  if (r.kind === "ride" && r.vehicle) {
    const v = r.vehicle;
    const days = span + 1;
    const chauffeured = r.drive_type === "chauffeured" || (r.drive_type === null && r.with_driver);
    const rate = chauffeured
      ? (v.chauffeur_price_per_day ?? (v.price_per_day ?? 0) + v.driver_fee_per_day)
      : (v.price_per_day ?? 0);
    return {
      amount: days * rate,
      note: `${days} day${days === 1 ? "" : "s"}, ${chauffeured ? "chauffeured" : "self-drive"}`,
      currency: v.currency,
    };
  }
  if (r.kind === "tour" && r.tour) {
    const t = r.tour;
    if (t.price_basis === "per_person") {
      return { amount: r.guests * t.price, note: `${r.guests} ${r.guests === 1 ? "person" : "people"}`, currency: t.currency };
    }
    if (t.price_basis === "per_day") {
      const days = span + 1;
      return { amount: days * t.price, note: `${days} day${days === 1 ? "" : "s"}`, currency: t.currency };
    }
    return { amount: t.price, note: "per group", currency: t.currency };
  }
  return null;
}

  const r = data as unknown as Req;

  const title = r.property?.title ?? r.vehicle?.title ?? r.tour?.title ?? "a removed listing";

    (r.final_total !== null ? ` The total for your booking is ${money(r.final_total, currency)}.` : "");

import type { Metadata } from "next";
import Link from "next/link";
import Icon, { type IconName } from "@/components/Icon";
import RequestList, { type RequestRow } from "@/components/admin/RequestList";
import { supabaseServer } from "@/lib/supabase/server";

export const metadata: Metadata = { title: "Requests" };

const STATUS_TABS = ["all", "new", "contacted", "confirmed", "declined", "cancelled", "completed"];
const KIND_TABS: [string, string][] = [
  ["all", "All services"],
  ["stay", "Stays"],
  ["ride", "Rides"],
  ["tour", "Tours"],
];
const KIND_ICON: Record<string, IconName> = { stay: "home", ride: "car", tour: "compass" };

function href(status: string, kind: string) {
  const p = new URLSearchParams();
  if (status !== "all") p.set("status", status);
  if (kind !== "all") p.set("kind", kind);
  const q = p.toString();
  return q ? `/admin/requests?${q}` : "/admin/requests";
}

export default async function RequestsPage({
  searchParams,
}: {
  searchParams: Promise<{ status?: string; kind?: string }>;
}) {
  const { status: rawStatus, kind: rawKind } = await searchParams;
  const status = STATUS_TABS.includes(rawStatus ?? "") ? (rawStatus as string) : "all";
  const kind = KIND_TABS.some(([k]) => k === rawKind) ? (rawKind as string) : "all";

  const supabase = await supabaseServer();
  let q = supabase
    .from("booking_requests")
    .select(
      "id, reference, kind, starts_on, ends_on, full_name, status, created_at, drive_type, property:properties(title), vehicle:vehicles(title), tour:tours(title)"
    )
    .order("created_at", { ascending: false })
    .limit(200);
  if (status !== "all") q = q.eq("status", status);
  if (kind !== "all") q = q.eq("kind", kind);
  const { data } = await q;
  const rows = (data ?? []) as unknown as RequestRow[];

  return (
    <div className="space-y-8">
      <div>
        <h1 className="text-3xl">requests</h1>
        <p className="mt-2 text-sm text-muted">Everything guests have sent, newest first.</p>
      </div>

      <div className="space-y-3">
        <nav aria-label="Filter by service" className="-mx-5 flex gap-x-6 overflow-x-auto whitespace-nowrap px-5 text-sm sm:mx-0 sm:flex-wrap sm:px-0">
          {KIND_TABS.map(([k, label]) => (
            <Link
              key={k}
              href={href(status, k)}
              aria-current={k === kind ? "page" : undefined}
              className={`nav-link inline-flex items-center gap-2 tracking-wider ${k === kind ? "" : "text-muted"}`}
            >
              {k !== "all" && <Icon name={KIND_ICON[k]} size={15} />}
              {label}
            </Link>
          ))}
        </nav>
        <nav aria-label="Filter by status" className="-mx-5 flex gap-x-6 overflow-x-auto whitespace-nowrap border-b border-line px-5 pb-3 text-sm sm:mx-0 sm:flex-wrap sm:px-0">
          {STATUS_TABS.map((t) => (
            <Link
              key={t}
              href={href(t, kind)}
              aria-current={t === status ? "page" : undefined}
              className={`nav-link capitalize tracking-wider ${t === status ? "" : "text-muted"}`}
            >
              {t}
            </Link>
          ))}
        </nav>
      </div>

      {rows.length === 0 ? (
        <p className="text-sm text-muted">No requests match these filters.</p>
      ) : (
        <RequestList rows={rows} />
      )}
    </div>
  );
}

// Guest app: /pwa-icon/192, /512, /maskable
// Admin app: /pwa-icon/admin-180, /admin-192, /admin-512, /admin-maskable
export async function GET(_req: Request, { params }: { params: Promise<{ size: string }> }) {
  const { size } = await params;
  const options: Record<string, { px: number; ratio: number; admin?: boolean }> = {
    "192": { px: 192, ratio: 0.62 },
    "512": { px: 512, ratio: 0.62 },
    maskable: { px: 512, ratio: 0.5 },
    "admin-180": { px: 180, ratio: 0.62, admin: true },
    "admin-192": { px: 192, ratio: 0.62, admin: true },
    "admin-512": { px: 512, ratio: 0.62, admin: true },
    "admin-maskable": { px: 512, ratio: 0.5, admin: true },
  };
  const o = options[size];
  if (!o) return new Response("Not found", { status: 404 });

  return new ImageResponse(<BrandIcon size={o.px} ratio={o.ratio} admin={o.admin} />, {
    width: o.px,
    height: o.px,
    headers: { "Cache-Control": "public, max-age=31536000, immutable" },
  });
}

export default async function TourPage({ params, searchParams }: Props) {
  const { slug } = await params;
  const sp = await searchParams;
  const tour = await getTourDetail(slug);
  if (!tour) notFound();

  const [fullDays, policies] = await Promise.all([
    getFullDays("tour", tour.id, tour.capacity_per_day),
    getPolicies("tour"),
  ]);

  const guide = tour.tour_type === "guide";
  const today = new Date().toLocaleDateString("en-CA", { timeZone: "Africa/Nairobi" });
  const iso = /^\d{4}-\d{2}-\d{2}$/;
  const from = sp.from && iso.test(sp.from) && sp.from >= today ? sp.from : "";
  const to = guide && from && sp.to && iso.test(sp.to) && sp.to >= from ? sp.to : "";

  const images = sortImages(tour.listing_images);
  const sub = [tour.location?.name, tour.duration_text].filter(Boolean).join(", ");

  const facts: [string, string][] = [];
  if (tour.location) facts.push(["Location", tour.location.name]);
  facts.push(["Type", guide ? "Guide hired by the day" : "Fixed tour"]);
  if (tour.duration_text) facts.push(["Duration", tour.duration_text]);
  if (tour.meeting_point) facts.push(["Meeting point", tour.meeting_point]);
  facts.push([
    "Group size",
    tour.min_people === tour.max_group ? plural(tour.max_group, "person") : `${tour.min_people} to ${tour.max_group} people`,

"use client";

import { useEffect, useState } from "react";

type InstallEvent = Event & {
  prompt: () => Promise<void>;
  userChoice: Promise<{ outcome: string }>;
};

const button =
  "border border-gold/60 px-5 py-2.5 text-sm tracking-widest text-gold transition-colors hover:bg-gold/10";

export default function InstallApp() {
  const [deferred, setDeferred] = useState<InstallEvent | null>(null);
  const [ios, setIos] = useState(false);
  const [installed, setInstalled] = useState(false);
  const [hint, setHint] = useState(false);

  useEffect(() => {
    const standalone =
      window.matchMedia("(display-mode: standalone)").matches ||
      (navigator as Navigator & { standalone?: boolean }).standalone === true;
    setInstalled(standalone);
    setIos(/iphone|ipad|ipod/i.test(navigator.userAgent) && !standalone);

    const onPrompt = (e: Event) => {
      e.preventDefault();
      setDeferred(e as InstallEvent);
    };
    const onInstalled = () => {
      setInstalled(true);
      setDeferred(null);
    };
    window.addEventListener("beforeinstallprompt", onPrompt);
    window.addEventListener("appinstalled", onInstalled);
    return () => {
      window.removeEventListener("beforeinstallprompt", onPrompt);
      window.removeEventListener("appinstalled", onInstalled);
    };
  }, []);

  if (installed) return null;

  if (deferred) {
    return (
      <button
        type="button"
        className={button}
        onClick={async () => {
          await deferred.prompt();
          await deferred.userChoice;
          setDeferred(null);
        }}
      >
        Install the app
      </button>
    );
  }

  if (ios) {
    return (
      <div>
        <button type="button" className={button} aria-expanded={hint} onClick={() => setHint((h) => !h)}>
          Install the app
        </button>
        {hint && (
          <p className="mt-3 max-w-xs text-xs text-cream/70">
            On iPhone: tap the Share button in Safari, then choose Add to Home Screen.
          </p>
        )}
      </div>
    );
  }

  return null;
}

    if (process.env.NODE_ENV !== "production") {
      navigator.serviceWorker.getRegistrations().then((list) => list.forEach((r) => r.unregister()));
      if ("caches" in window) {
        caches.keys().then((keys) => keys.filter((k) => k.startsWith("lumora-")).forEach((k) => caches.delete(k)));
      }
      return;
    }

    navigator.serviceWorker.register("/sw.js", { scope: "/" }).catch(() => {});
  }, []);
  return null;
}

"use client";

import { useEffect, useId, useRef, useState } from "react";

type Props = {
  id: string;
  name: string;
  options: [string, string][];
  defaultValue?: string;
  className?: string;
  onChange?: (value: string) => void;
};

export default function ThemedSelect({ id, name, options, defaultValue, className = "", onChange }: Props) {
  const listId = useId();
  const root = useRef<HTMLDivElement>(null);
  const first = options[0]?.[0] ?? "";
  const [value, setValue] = useState(
    defaultValue && options.some(([v]) => v === defaultValue) ? defaultValue : first
  );
  const [open, setOpen] = useState(false);
  const [active, setActive] = useState(0);

  const selectedIndex = Math.max(0, options.findIndex(([v]) => v === value));
  const label = options[selectedIndex]?.[1] ?? "";

  useEffect(() => {
    if (!open) return;
    const close = (e: MouseEvent) => {
      if (!root.current?.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener("mousedown", close);
    return () => document.removeEventListener("mousedown", close);
  }, [open]);

  useEffect(() => {
    if (open) document.getElementById(`${listId}-${active}`)?.scrollIntoView({ block: "nearest" });
  }, [open, active, listId]);

  function openList() {
    setActive(selectedIndex);
    setOpen(true);
  }

  function choose(i: number) {
    const v = options[i][0];
    setValue(v);
    setOpen(false);
    onChange?.(v);
  }

  function onKeyDown(e: React.KeyboardEvent<HTMLButtonElement>) {
    switch (e.key) {
      case "ArrowDown":
        e.preventDefault();
        if (!open) openList();
        else setActive((a) => Math.min(a + 1, options.length - 1));
        break;
      case "ArrowUp":
        e.preventDefault();
        if (!open) openList();
        else setActive((a) => Math.max(a - 1, 0));
        break;
      case "Home":
        if (open) {
          e.preventDefault();
          setActive(0);
        }
        break;
      case "End":
        if (open) {
          e.preventDefault();
          setActive(options.length - 1);
        }
        break;
      case "Enter":
      case " ":
        e.preventDefault();
        if (open) choose(active);
        else openList();
        break;
      case "Escape":
        if (open) {
          e.preventDefault();
          setOpen(false);
        }
        break;
      case "Tab":
        setOpen(false);
        break;
    }
  }

  return (
    <div ref={root} className="relative">
      <input type="hidden" name={name} value={value} />
      <button
        type="button"
        id={id}
        role="combobox"
        aria-haspopup="listbox"
        aria-expanded={open}
        aria-controls={listId}
        aria-activedescendant={open ? `${listId}-${active}` : undefined}
        onClick={() => (open ? setOpen(false) : openList())}
        onKeyDown={onKeyDown}
        onKeyUp={(e) => {
          if (e.key === " ") e.preventDefault();
        }}
        className={`${className} flex items-center justify-between gap-3 text-left`}
      >
        <span>{label}</span>
        <svg
          width="12" height="8" viewBox="0 0 12 8" fill="none" stroke="currentColor" strokeWidth="1.5"
          strokeLinecap="round" strokeLinejoin="round" aria-hidden="true"
          className={`shrink-0 text-accent-strong transition-transform duration-200 ${open ? "rotate-180" : ""}`}
        >
          <path d="M1 1.5l5 5 5-5" />
        </svg>
      </button>

      {open && (
        <ul
          id={listId}
          role="listbox"
          className="absolute z-30 mt-1 max-h-60 w-full overflow-auto border border-accent bg-bg py-1 text-sm text-fg shadow-lg"
        >
          {options.map(([v, l], i) => (
            <li
              key={v}
              id={`${listId}-${i}`}
              role="option"
              aria-selected={v === value}
              onMouseEnter={() => setActive(i)}
              onMouseDown={(e) => {
                e.preventDefault();
                choose(i);
              }}
              className={`cursor-pointer px-3 py-2.5 ${
                i === active ? "bg-accent text-on-accent" : ""
              } ${v === value ? "tracking-wide" : ""}`}
            >
              {l}
            </li>
          ))}
        </ul>
      )}
    </div>
  );
}

import { useEffect, useRef } from "react";

/**
 * A form section. On a computer it is always open. On a phone it is a card that opens and closes.
 * openOnPhone keeps a section open at first, for sections with required fields.
 */
export default function FormSection({
  title,
  openOnPhone = false,
  children,
}: {
  title: string;
  openOnPhone?: boolean;
  children: React.ReactNode;
}) {
  const ref = useRef<HTMLDetailsElement>(null);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const wide = window.matchMedia("(min-width: 768px)");
    if (!wide.matches && !openOnPhone) el.open = false;
    // if the screen becomes wide later, make sure nothing stays hidden
    const sync = () => {
      if (wide.matches) el.open = true;
    };
    wide.addEventListener("change", sync);
    return () => wide.removeEventListener("change", sync);
  }, [openOnPhone]);

  return (
    <details ref={ref} open className="group border border-line md:border-0">
      <summary className="flex cursor-pointer list-none items-center justify-between gap-4 p-4 md:pointer-events-none md:cursor-default md:border-b md:border-line md:p-0 md:pb-2 [&::-webkit-details-marker]:hidden">

      </summary>

type Kind = "stay" | "ride" | "tour";

  listing_images: { path: string; sort_order: number; is_cover: boolean }[];

const CFG: Record<Kind, { table: string; base: string; heading: string; intro: string; add: string; cols: string; empty: string }> = {
  stay: { table: "properties", base: "/admin/stays", heading: "stays", intro: "Your apartments.", add: "Add a stay", cols: "price_per_night", empty: "stay" },
  ride: { table: "vehicles", base: "/admin/rides", heading: "rides", intro: "Your cars.", add: "Add a car", cols: "price_per_day, chauffeur_price_per_day", empty: "car" },
  tour: { table: "tours", base: "/admin/tours", heading: "tours", intro: "Your tours and guides.", add: "Add a tour", cols: "price, price_basis", empty: "tour" },
};

const basisLabel: Record<string, string> = { per_person: "per person", per_group: "per group", per_day: "per day" };

function priceText(kind: Kind, r: Row) {
  if (kind === "stay") return `${money(r.price_per_night ?? 0, r.currency)} per night`;
  if (kind === "ride") {
    if (r.price_per_day != null) return `${money(r.price_per_day, r.currency)} per day`;
    return `${money(r.chauffeur_price_per_day ?? 0, r.currency)} per day, chauffeured`;
  }
  return `${money(r.price ?? 0, r.currency)} ${basisLabel[r.price_basis ?? "per_person"]}`;
}

function coverOf(images: Row["listing_images"]) {
  const sorted = [...(images ?? [])].sort((a, b) => Number(b.is_cover) - Number(a.is_cover) || a.sort_order - b.sort_order);
  return sorted[0]?.path ?? null;
}

export default async function ListingIndex({ kind }: { kind: Kind }) {
  const supabase = await supabaseServer();
  const c = CFG[kind];
  const { data } = await supabase
    .from(c.table)
    .select(`id, title, status, featured, currency, location:locations(name), listing_images(path, sort_order, is_cover), ${c.cols}`)
    .order("sort_order")
    .order("created_at", { ascending: false });
  const raw = (data ?? []) as unknown as Row[];

  const rows: ListingRow[] = raw.map((r) => ({
    id: r.id,
    title: r.title,
    status: r.status,
    featured: r.featured,
    location: r.location?.name ?? "No location",
    price: priceText(kind, r),
    photos: r.listing_images?.length ?? 0,
    cover: coverOf(r.listing_images),
  }));

  return (
    <div className="space-y-8">
      <div className="flex flex-wrap items-end justify-between gap-4">

  amount: number | string;
  method: string;

  paid_on: string;

export default async function PaymentsPanel({ requestId }: { requestId: string }) {
  const sb = await supabaseServer();
  const [{ data: req }, { data: rows }] = await Promise.all([
    sb.from("booking_requests").select("final_total, estimate_total").eq("id", requestId).maybeSingle(),
    sb
      .from("payments")
      .select("id, kind, amount, method, reference, paid_on, note")
      .eq("request_id", requestId)
      .order("paid_on")
      .order("created_at"),
  ]);

          <dd>{money(net, "KES")}</dd>

import { useRef, useState } from "react";
import { supabaseBrowser } from "@/lib/supabase/client";
import { imageUrl } from "@/lib/format";
import { refreshPublic } from "@/app/admin/(panel)/listing-actions";

type Img = { id: string; path: string; alt: string | null; sort_order: number; is_cover: boolean };

// Shrinks big phone photos before upload so pages stay fast.
async function shrink(file: File): Promise<Blob> {
  try {
    const bmp = await createImageBitmap(file);
    const scale = Math.min(1, 2200 / Math.max(bmp.width, bmp.height));
    const c = document.createElement("canvas");
    c.width = Math.round(bmp.width * scale);
    c.height = Math.round(bmp.height * scale);
    c.getContext("2d")!.drawImage(bmp, 0, 0, c.width, c.height);
    const blob: Blob | null = await new Promise((r) => c.toBlob(r, "image/jpeg", 0.85));
    return blob ?? file;
  } catch {
    return file;
  }
}

const btn =
  "border border-line px-3 py-1.5 text-xs tracking-wider text-muted transition-colors hover:border-accent hover:text-fg disabled:opacity-40";

export default function PhotoManager({
  kind,
  listingId,
  initial,
}: {
  kind: "stay" | "ride" | "tour";
  listingId: string;
  initial: Img[];
}) {
  const [images, setImages] = useState<Img[]>(initial);
  const [busy, setBusy] = useState(false);
  const [error, setError] = useState("");
  const input = useRef<HTMLInputElement>(null);
  const col = kind === "stay" ? "property_id" : kind === "ride" ? "vehicle_id" : "tour_id";
  const folder = kind === "stay" ? "properties" : kind === "ride" ? "vehicles" : "tours";
  const sorted = [...images].sort((a, b) => a.sort_order - b.sort_order);

  async function onFiles(files: FileList | null) {
    if (!files || files.length === 0) return;
    setBusy(true);
    setError("");
    const sb = supabaseBrowser();
    let list = [...images];
    let order = Math.max(-1, ...list.map((i) => i.sort_order)) + 1;

    for (const f of Array.from(files)) {
      if (!/^image\/(jpeg|png|webp)$/.test(f.type)) {
        setError(`${f.name}: please use a JPG, PNG or WebP photo.`);
        continue;
      }
      const blob = await shrink(f);
      if (blob.size > 8 * 1024 * 1024) {
        setError(`${f.name} is too large (maximum 8 MB).`);
        continue;
      }
      const ext = blob.type === "image/png" ? "png" : blob.type === "image/webp" ? "webp" : "jpg";
      const path = `${folder}/${listingId}/${crypto.randomUUID()}.${ext}`;
      const up = await sb.storage
        .from("listing-images")
        .upload(path, blob, { contentType: blob.type || f.type, cacheControl: "31536000" });
      if (up.error) {
        setError(`Could not upload ${f.name}. Please try again.`);
        continue;
      }
      const ins = await sb
        .from("listing_images")
        .insert({ [col]: listingId, path, alt: null, sort_order: order++, is_cover: list.length === 0 })
        .select("id, path, alt, sort_order, is_cover")
        .single();
      if (ins.error || !ins.data) {
        await sb.storage.from("listing-images").remove([path]);
        setError(`Could not save ${f.name}. Please try again.`);
        continue;
      }
      list = [...list, ins.data as Img];
      setImages(list);
    }
    setBusy(false);
    if (input.current) input.current.value = "";
    await refreshPublic();
  }

  async function makeCover(id: string) {
    const sb = supabaseBrowser();
    await sb.from("listing_images").update({ is_cover: false }).eq(col, listingId);
    const { error: e } = await sb.from("listing_images").update({ is_cover: true }).eq("id", id);
    if (e) return setError("Could not change the cover photo.");
    setImages((p) => p.map((i) => ({ ...i, is_cover: i.id === id })));
    await refreshPublic();
  }

  async function move(id: string, dir: -1 | 1) {
    const i = sorted.findIndex((x) => x.id === id);
    const j = i + dir;
    if (i < 0 || j < 0 || j >= sorted.length) return;
    const a = sorted[i];
    const b = sorted[j];
    const sb = supabaseBrowser();
    const [r1, r2] = await Promise.all([
      sb.from("listing_images").update({ sort_order: b.sort_order }).eq("id", a.id),
      sb.from("listing_images").update({ sort_order: a.sort_order }).eq("id", b.id),
    ]);
    if (r1.error || r2.error) return setError("Could not reorder the photos.");
    setImages((p) =>
      p.map((x) => (x.id === a.id ? { ...x, sort_order: b.sort_order } : x.id === b.id ? { ...x, sort_order: a.sort_order } : x))
    );
    await refreshPublic();
  }

  async function saveAlt(id: string, text: string) {
    const value = text.trim() || null;
    const current = images.find((i) => i.id === id);
    if (!current || (current.alt ?? null) === value) return;
    const { error: e } = await supabaseBrowser().from("listing_images").update({ alt: value }).eq("id", id);
    if (e) return setError("Could not save the description.");
    setImages((p) => p.map((i) => (i.id === id ? { ...i, alt: value } : i)));
    await refreshPublic();
  }

  async function remove(img: Img) {
    if (!confirm("Remove this photo?")) return;
    const sb = supabaseBrowser();
    await sb.storage.from("listing-images").remove([img.path]);
    const { error: e } = await sb.from("listing_images").delete().eq("id", img.id);
    if (e) return setError("Could not remove the photo.");
    const rest = images.filter((i) => i.id !== img.id);
    setImages(rest);
    if (img.is_cover && rest.length > 0) {
      const first = [...rest].sort((a, b) => a.sort_order - b.sort_order)[0];
      await makeCover(first.id);
    }
    await refreshPublic();
  }

  return (
    <section className="space-y-5" aria-labelledby="photos">
      <h2 id="photos" className="border-b border-line pb-2 text-xl">photos</h2>
      <p className="text-sm text-muted">
        The cover photo is shown first on the website. Use JPG, PNG or WebP. Large photos are shrunk for you.
      </p>

      <div>
        <input
          ref={input} id="photo-input" type="file" multiple accept="image/jpeg,image/png,image/webp"
          className="sr-only" disabled={busy} onChange={(e) => onFiles(e.target.files)}
        />
        <label
          htmlFor="photo-input"
          className={`inline-block cursor-pointer bg-accent px-6 py-3 text-sm tracking-widest text-on-accent transition-opacity hover:opacity-90 ${busy ? "pointer-events-none opacity-50" : ""}`}
        >
          {busy ? "Uploading..." : "Add photos"}
        </label>
      </div>

      {error && <p role="alert" className="text-sm underline decoration-accent underline-offset-4">{error}</p>}

      {sorted.length === 0 ? (
        <p className="text-sm text-muted">No photos yet. Add the best one first; it becomes the cover.</p>
      ) : (
        <ul className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
          {sorted.map((img, n) => (
            <li key={img.id} className="border border-line">
              <div className="relative aspect-[4/3] bg-surface">
                {/* eslint-disable-next-line @next/next/no-img-element */}
                <img src={imageUrl(img.path)} alt={img.alt ?? ""} className="h-full w-full object-cover" />
                {img.is_cover && (
                  <span className="absolute left-2 top-2 bg-accent px-2 py-0.5 text-xs tracking-wider text-on-accent">Cover</span>
                )}

              <div className="space-y-3 p-3">

                  aria-label="Photo description"
                  placeholder="Describe the photo (helps accessibility)"
                  onBlur={(e) => saveAlt(img.id, e.target.value)}
                  className="w-full border border-line bg-transparent px-2.5 py-2 text-xs focus:border-accent focus:outline-none"
                />
                <div className="flex flex-wrap gap-2">
                  {!img.is_cover && <button type="button" className={btn} onClick={() => makeCover(img.id)}>Make cover</button>}
                  <button type="button" className={btn} disabled={n === 0} onClick={() => move(img.id, -1)}>Earlier</button>
                  <button type="button" className={btn} disabled={n === sorted.length - 1} onClick={() => move(img.id, 1)}>Later</button>
                  <button type="button" className={btn} onClick={() => remove(img)}>Remove</button>
                </div>
              </div>
            </li>
          ))}
        </ul>
      )}
    </section>

function keyToBytes(base64url: string) {
  const pad = "=".repeat((4 - (base64url.length % 4)) % 4);
  const raw = atob((base64url + pad).replace(/-/g, "+").replace(/_/g, "/"));
  const out = new Uint8Array(raw.length);
  for (let i = 0; i < raw.length; i++) out[i] = raw.charCodeAt(i);
  return out;
}

function deviceLabel() {
  const ua = navigator.userAgent;
  const os = /android/i.test(ua) ? "Android" : /iphone|ipad|ipod/i.test(ua) ? "iPhone" : /windows/i.test(ua) ? "Windows" : /mac/i.test(ua) ? "Mac" : "Device";
  const app = /edg/i.test(ua) ? "Edge" : /firefox/i.test(ua) ? "Firefox" : /chrome|crios/i.test(ua) ? "Chrome" : /safari/i.test(ua) ? "Safari" : "Browser";
  return `${os}, ${app}`;
}

export default function PushControls({
  publicKey,
  serverReady,
  devices,
}: {
  publicKey: string | null;
  serverReady: boolean;
  devices: Device[];
}) {
  const [support, setSupport] = useState<Support>("checking");
  const [permission, setPermission] = useState<NotificationPermission>("default");
  const [here, setHere] = useState<string | null>(null);
  const [busy, setBusy] = useState(false);
  const [msg, setMsg] = useState<{ text: string; bad?: boolean } | null>(null);

  useEffect(() => {
    let alive = true;
    (async () => {
      const ios = /iphone|ipad|ipod/i.test(navigator.userAgent);
      const standalone =
        window.matchMedia("(display-mode: standalone)").matches ||
        (navigator as Navigator & { standalone?: boolean }).standalone === true;
      const capable = "serviceWorker" in navigator && "PushManager" in window && "Notification" in window;
      if (!capable) {
        if (alive) setSupport(ios && !standalone ? "ios-install" : "unsupported");
        return;
      }
      if (alive) setPermission(Notification.permission);
      const reg = await navigator.serviceWorker.getRegistration();
      if (!alive) return;
      if (!reg) {
        setSupport("no-worker");
        return;
      }
      const sub = await reg.pushManager.getSubscription();
      if (!alive) return;
      setHere(sub?.endpoint ?? null);
      setSupport("ready");
    })();
    return () => {
      alive = false;
    };
  }, []);

  async function turnOn() {
    if (!publicKey) return;
    setBusy(true);
    setMsg(null);
    try {
      const perm = await Notification.requestPermission();
      setPermission(perm);
      if (perm !== "granted") {
        setMsg({
          text: "Notifications are blocked. Allow them for Lumora in your phone or browser settings, then try again.",
          bad: true,
        });
        return;
      }
      const reg = await navigator.serviceWorker.ready;
      const old = await reg.pushManager.getSubscription();
      if (old) {
        await removeSubscription(old.endpoint);
        await old.unsubscribe();
      }
      const sub = await reg.pushManager.subscribe({ userVisibleOnly: true, applicationServerKey: keyToBytes(publicKey) });
      const j = sub.toJSON();
      const r = await saveSubscription({
        endpoint: sub.endpoint,
        p256dh: j.keys?.p256dh ?? "",
        auth: j.keys?.auth ?? "",
        label: deviceLabel(),
      });
      if (r.error) {
        setMsg({ text: r.error, bad: true });
        return;
      }
      setHere(sub.endpoint);
      setMsg({ text: "Notifications are on for this phone. Send a test to check." });
    } catch {
      setMsg({ text: "Could not turn on notifications. Please try again.", bad: true });
    } finally {
      setBusy(false);
    }
  }

  async function turnOff() {
    setBusy(true);
    setMsg(null);
    try {
      const reg = await navigator.serviceWorker.getRegistration();
      const sub = await reg?.pushManager.getSubscription();
      if (sub) {
        await removeSubscription(sub.endpoint);
        await sub.unsubscribe();
      }
      setHere(null);
      setMsg({ text: "Notifications are off for this phone." });
    } catch {
      setMsg({ text: "Could not turn them off. Please try again.", bad: true });
    } finally {
      setBusy(false);
    }
  }

  async function test() {
    setBusy(true);
    setMsg(null);
    const r = await sendTest();
    setMsg(r.error ? { text: r.error, bad: true } : { text: r.message ?? "Sent." });
    setBusy(false);
  }

    await removeSubscription(endpoint);
    setBusy(false);
  }

    <div className="space-y-8">

      )}

      {support === "checking" && <p className="text-sm text-muted">Checking this phone...</p>}

      {support === "ios-install" && (
        <p className="border border-line p-4 text-sm">
          On an iPhone, notifications only work after the Lumora Admin app is on your Home Screen. Open this page in
          Safari, tap Share, choose Add to Home Screen, then open the app from your Home Screen and come back here.
        </p>
      )}

      {support === "unsupported" && (
        <p className="border border-line p-4 text-sm">
          This browser cannot receive notifications. Use Chrome on Android, or Safari on an iPhone with the app added
          to the Home Screen.
        </p>
      )}

      {support === "no-worker" && (
        <p className="border border-line p-4 text-sm">
          The app&apos;s background helper is not running yet. Reload this page once. Notifications only work on the
          live website, not on the test version on your computer.
        </p>
      )}

      {support === "ready" && (
        <div className="space-y-5">
          <p className="flex items-center gap-3 text-sm">
            <Icon name="bell" size={20} />
            This phone:{" "}
            <span className="tracking-wide">
              {permission === "denied" ? "blocked" : on ? "notifications on" : "notifications off"}
            </span>
          </p>
          <div className="flex flex-wrap gap-3">
            {on ? (
              <>
                <button type="button" onClick={test} disabled={busy} className={primary}>
                  Send a test notification
                </button>
                <button type="button" onClick={turnOff} disabled={busy} className={quiet}>
                  Turn off on this phone
                </button>
              </>
            ) : (
              <button type="button" onClick={turnOn} disabled={busy || !publicKey} className={primary}>
                Turn on notifications
              </button>
            )}
          </div>
        </div>
      )}

      {msg && (
        <p
          role={msg.bad ? "alert" : "status"}
          className={`text-sm ${msg.bad ? "underline decoration-accent underline-offset-4" : "text-muted"}`}
        >
          {msg.text}

const LocationPicker = dynamic(() => import("./LocationPicker"), {

export type DeliveryMode = "none" | "point" | "custom";
export type DeliveryPoint = { id: string; name: string; kind: "airport" | "sgr" | "other"; fee: number };
export type DeliveryValue = { mode: DeliveryMode; pointId: string; time: string; details: string; pin: Pin | null };

const box =
  "w-full border border-line bg-transparent px-3 py-2.5 text-sm text-fg placeholder:text-muted focus:border-accent focus:outline-none";
const lab = "mb-1.5 block text-xs tracking-widest text-muted";

type Props = {
  name: "pickup" | "dropoff";
  title: string;
  timeLabel: string;
  labels: Record<DeliveryMode, string>;
  points: DeliveryPoint[];
  value: DeliveryValue;
  onChange: (v: DeliveryValue) => void;
  currency: string;
};

export default function DeliveryBlock({ name, title, timeLabel, labels, points, value, onChange, currency }: Props) {
  const point = points.find((p) => p.id === value.pointId) ?? points[0];
  const modes: DeliveryMode[] = points.length > 0 ? ["none", "point", "custom"] : ["none", "custom"];
  const pin = value.pin;

  return (
    <fieldset className="border-t border-line pt-4">
      <legend className="sr-only">{title}</legend>
      <p className="mb-3 text-xs tracking-widest text-muted">{title}</p>

      <div className="space-y-2.5">
        {modes.map((m) => (
          <label key={m} className="flex items-center gap-3 text-sm">
            <input
              type="radio"
              name={`${name}_choice`}
              checked={value.mode === m}
              onChange={() => onChange({ ...value, mode: m, pointId: value.pointId || points[0]?.id || "" })}
              className="h-4 w-4 accent-[var(--accent)]"
            />
            {labels[m]}
          </label>
        ))}
      </div>

      <input type="hidden" name={`${name}_mode`} value={value.mode} />
      {value.mode === "point" && <input type="hidden" name={`${name}_point_id`} value={point?.id ?? ""} />}
      {value.mode === "custom" && pin && (
        <>
          <input type="hidden" name={`${name}_lat`} value={pin.lat} />
          <input type="hidden" name={`${name}_lng`} value={pin.lng} />
        </>
      )}

      {value.mode === "point" && point && (
        <div className="mt-4">
          <label htmlFor={`${name}-where`} className={lab}>Where</label>
          <ThemedSelect
            id={`${name}-where`}
            name={`${name}_where`}
            defaultValue={point.id}

            className={box}
            options={points.map((p): [string, string] => [p.id, p.name])}
          />

        </div>
      )}

      {value.mode === "custom" && (
        <div className="mt-4 space-y-4">
          <LocationPicker value={pin} onChange={(p) => onChange({ ...value, pin: p })} />
          {pin && (
            <div>

              <input
                id={`${name}-address`}
                name={`${name}_address`}
                type="text"
                maxLength={300}
                value={pin.address}
                onChange={(e) => onChange({ ...value, pin: { ...pin, address: e.target.value } })}
                className={box}
                placeholder="Hotel name, gate or house number"
              />
            </div>
          )}
          <p className="text-xs text-muted">
            We will confirm the delivery price with you on WhatsApp before anything is charged.
          </p>
        </div>

        <div className="mt-4 grid grid-cols-2 gap-3">

            <input
              id={`${name}-time`}
              name={`${name}_time`}
              type="time"
              required
              value={value.time}
              onChange={(e) => onChange({ ...value, time: e.target.value })}
              className={box}
            />
          </div>
          <div>
            <label htmlFor={`${name}-details`} className={lab}>

            <input
              id={`${name}-details`}
              name={`${name}_flight`}
              type="text"
              maxLength={80}
              value={value.details}
              onChange={(e) => onChange({ ...value, details: e.target.value })}
              className={box}
              placeholder={value.mode === "point" ? "KQ 612" : "Gate B, ask for Sam"}
            />
          </div>
        </div>
      )}
    </fieldset>
  );
}

"use client";

import { useEffect, useRef, useState } from "react";
import L from "leaflet";
import "leaflet/dist/leaflet.css";
import Icon from "@/components/Icon";

export type Pin = { lat: number; lng: number; address: string };
type Place = { lat: string; lon: string; display_name: string };

const NOMINATIM = "https://nominatim.openstreetmap.org";
const MOMBASA: [number, number] = [-4.0435, 39.6682];

const pinIcon = L.divIcon({
  className: "",
  html: '<svg width="34" height="34" viewBox="0 0 34 34" aria-hidden="true"><circle cx="17" cy="17" r="11" fill="#FBF8F3" stroke="#1B2A41" stroke-width="5"/><circle cx="17" cy="17" r="4" fill="#1B2A41"/></svg>',
  iconSize: [34, 34],
  iconAnchor: [17, 17],
});

const box =
  "w-full border border-line bg-transparent px-3 py-2.5 text-sm text-fg placeholder:text-muted focus:border-accent focus:outline-none";

async function reverse(lat: number, lng: number): Promise<string> {
  try {
    const r = await fetch(`${NOMINATIM}/reverse?format=jsonv2&zoom=18&accept-language=en&lat=${lat}&lon=${lng}`);
    if (!r.ok) return "";
    const j = await r.json();
    return typeof j.display_name === "string" ? j.display_name.slice(0, 250) : "";
  } catch {
    return "";
  }
}

export default function LocationPicker({
  value,
  onChange,
}: {
  value: Pin | null;
  onChange: (p: Pin | null) => void;
}) {
  const el = useRef<HTMLDivElement>(null);
  const map = useRef<L.Map | null>(null);
  const marker = useRef<L.Marker | null>(null);
  const seq = useRef(0);
  const change = useRef(onChange);
  const [query, setQuery] = useState("");
  const [results, setResults] = useState<Place[]>([]);
  const [busy, setBusy] = useState(false);
  const [note, setNote] = useState("");

  useEffect(() => {
    change.current = onChange;
  }, [onChange]);

  function put(lat: number, lng: number, address?: string) {
    const la = Math.round(lat * 1e6) / 1e6;
    const lo = Math.round(lng * 1e6) / 1e6;
    if (marker.current) marker.current.setLatLng([la, lo]);
    else if (map.current) {
      const m = L.marker([la, lo], { icon: pinIcon, draggable: true }).addTo(map.current);
      m.on("dragend", () => {
        const p = m.getLatLng();
        put(p.lat, p.lng);
      });
      marker.current = m;
    }
    const mine = ++seq.current;
    change.current({ lat: la, lng: lo, address: address ?? "" });
    if (address === undefined) {
      reverse(la, lo).then((a) => {
        if (a && mine === seq.current) change.current({ lat: la, lng: lo, address: a });
      });
    }
  }

  useEffect(() => {
    if (!el.current || map.current) return;
    const m = L.map(el.current, {
      center: value ? [value.lat, value.lng] : MOMBASA,
      zoom: value ? 16 : 12,
      scrollWheelZoom: false,
    });
    L.tileLayer("https://tile.openstreetmap.org/{z}/{x}/{y}.png", {
      maxZoom: 19,
      attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors',
    }).addTo(m);
    m.on("click", (e: L.LeafletMouseEvent) => put(e.latlng.lat, e.latlng.lng));
    map.current = m;
    if (value) put(value.lat, value.lng, value.address);
    return () => {
      m.remove();
      map.current = null;
      marker.current = null;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  async function search() {
    const q = query.trim();
    if (q.length < 3) return setNote("Type at least 3 letters.");
    setBusy(true);
    setNote("");
    setResults([]);
    try {
      const r = await fetch(
        `${NOMINATIM}/search?format=jsonv2&limit=5&countrycodes=ke&accept-language=en&q=${encodeURIComponent(q)}`
      );
      const j: Place[] = await r.json();
      if (j.length === 0) setNote("Nothing found. Try a landmark or estate name, or tap the map.");
      setResults(j);
    } catch {
      setNote("Search is not available right now. Tap the map to place the pin.");
    }
    setBusy(false);
  }

  function choose(p: Place) {
    const la = Number(p.lat);
    const lo = Number(p.lon);
    map.current?.setView([la, lo], 16);
    put(la, lo, p.display_name.slice(0, 250));
    setResults([]);
  }

  function clear() {
    marker.current?.remove();
    marker.current = null;
    seq.current++;
    onChange(null);
  }

  return (
    <div className="space-y-3">
      <div className="flex gap-2">
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onKeyDown={(e) => {
            if (e.key === "Enter") {
              e.preventDefault();
              search();
            }
          }}
          placeholder="Search a place, hotel or estate"
          aria-label="Search for a place"
          className={box}
        />
        <button
          type="button"
          onClick={search}
          disabled={busy}
          className="inline-flex shrink-0 items-center gap-2 border border-accent px-4 text-xs tracking-widest transition-colors hover:bg-accent/15 disabled:opacity-50"
        >
          <Icon name="search" size={15} />
          {busy ? "Searching" : "Find"}
        </button>
      </div>

      {note && (
        <p role="status" className="text-xs text-muted">
          {note}
        </p>
      )}

      {results.length > 0 && (
        <ul className="divide-y divide-line border border-line text-sm">
          {results.map((p) => (
            <li key={`${p.lat},${p.lon},${p.display_name}`}>
              <button
                type="button"
                onClick={() => choose(p)}
                className="block w-full px-3 py-2.5 text-left transition-colors hover:bg-accent/15"
              >
                {p.display_name}
              </button>
            </li>
          ))}
        </ul>
      )}

      <div
        ref={el}
        className="relative z-0 h-64 w-full border border-line"
        role="application"
        aria-label="Map. Tap or click to place the pin."
      />

      <div className="flex flex-wrap items-center justify-between gap-3 text-xs text-muted">
        <span>{value ? "Drag the pin to adjust it." : "Tap the map to drop a pin."}</span>
        {value && (
          <button type="button" onClick={clear} className="underline decoration-line underline-offset-4 hover:decoration-accent">
            Remove pin
          </button>
        )}
      </div>
    </div>
  );
}

===== src/components/booking/RideRequestForm.tsx =====
"use client";

import { startTransition, useActionState, useEffect, useMemo, useState } from "react";
import { submitBooking, type BookingState } from "@/app/actions/booking";
import ThemedSelect from "@/components/ThemedSelect";
import { addDays, diffDays, eachDay, niceDate, toRanges } from "@/lib/dates";
import { money, plural } from "@/lib/format";
import ContactFields from "./ContactFields";
import DeliveryBlock, { type DeliveryPoint, type DeliveryValue } from "./DeliveryBlock";
import SuccessPanel from "./SuccessPanel";

type Drive = "self" | "chauffeur";

type Props = {
  listingId: string;
  title: string;
  currency: string;
  seats: number;
  minDays: number;
  selfPrice: number | null;
  chauffeurPrice: number | null;
  points: DeliveryPoint[];
  deliveryEnabled: boolean;
  fullDays: string[];
  initialFrom: string;
  initialTo: string;
  initialDrive: string;
  hasPolicies: boolean;
};

const box =
  "w-full border border-line bg-transparent px-3 py-2.5 text-sm text-fg placeholder:text-muted focus:border-accent focus:outline-none";
const lab = "mb-1.5 block text-xs tracking-widest text-muted";
const seg = (on: boolean) =>
  `px-4 py-2 text-sm transition-colors ${on ? "bg-accent text-on-accent" : "text-muted hover:text-fg"}`;

const EMPTY: DeliveryValue = { mode: "none", pointId: "", time: "", details: "", pin: null };

export default function RideRequestForm(p: Props) {
  const [state, action, pending] = useActionState<BookingState, FormData>(submitBooking, { ok: false });
  const both = p.selfPrice !== null && p.chauffeurPrice !== null;
  const [drive, setDrive] = useState<Drive>(
    p.initialDrive === "chauffeured" && p.chauffeurPrice !== null ? "chauffeur" : p.selfPrice !== null ? "self" : "chauffeur"
  );
  const [today, setToday] = useState("");
  const [start, setStart] = useState(p.initialFrom);
  const [end, setEnd] = useState(p.initialTo);
  const first = p.points[0]?.id ?? "";
  const [pickup, setPickup] = useState<DeliveryValue>({ ...EMPTY, pointId: first });
  const [dropoff, setDropoff] = useState<DeliveryValue>({ ...EMPTY, pointId: first });

  useEffect(() => setToday(new Date().toLocaleDateString("en-CA")), []);
  const full = useMemo(() => new Set(p.fullDays), [p.fullDays]);
  const ranges = useMemo(() => toRanges(p.fullDays), [p.fullDays]);

  const rate = (drive === "self" ? p.selfPrice : p.chauffeurPrice) ?? 0;
  const days = start && end ? diffDays(start, end) + 1 : null;

  let problem = "";
  if (start && end && days !== null) {
    if (days < 1) problem = "The return date cannot be before the pick-up date.";
    else if (days < p.minDays) problem = `This car needs at least ${plural(p.minDays, "day")}.`;
    else if (eachDay(start, addDays(end, 1)).some((d) => full.has(d))) {
      problem = "Those dates are not available. Please choose different dates.";
    }
  }

  const needsPin = (pickup.mode === "custom" && !pickup.pin) || (dropoff.mode === "custom" && !dropoff.pin);
  const feeOf = (v: DeliveryValue) =>
    v.mode === "point" ? (p.points.find((x) => x.id === v.pointId)?.fee ?? 0) : 0;
  const quoted = pickup.mode === "custom" || dropoff.mode === "custom";
  const pickupFee = feeOf(pickup);
  const dropoffFee = feeOf(dropoff);
  const base = days !== null && days >= 1 ? days * rate : null;
  const total = base !== null && !problem ? base + pickupFee + dropoffFee : null;

  function onSubmit(e: React.FormEvent<HTMLFormElement>) {
    e.preventDefault();
    const fd = new FormData(e.currentTarget);
    startTransition(() => action(fd));
  }

  if (state.ok && state.reference) {
    return (
      <SuccessPanel
        reference={state.reference}
        message={`It is for ${p.title}${start && end ? ` (${niceDate(start)} to ${niceDate(end)})` : ""}.`}
      />
    );
  }

  return (
    <form onSubmit={onSubmit} className="border border-line p-6" aria-labelledby="request-title">
      <p id="request-title" className="text-2xl tracking-wide">{money(rate, p.currency)}</p>
      <p className="text-sm text-muted">per day{drive === "chauffeur" ? ", with a chauffeur" : ""}</p>

      <input type="hidden" name="kind" value="ride" />
      <input type="hidden" name="listing_id" value={p.listingId} />
      <input type="hidden" name="drive_type" value={drive === "self" ? "self_drive" : "chauffeured"} />

      {both ? (
        <div role="group" aria-label="Drive type" className="mt-5 inline-flex border border-line">
          <button type="button" aria-pressed={drive === "self"} onClick={() => setDrive("self")} className={seg(drive === "self")}>
            Self-drive
          </button>
          <button type="button" aria-pressed={drive === "chauffeur"} onClick={() => setDrive("chauffeur")} className={seg(drive === "chauffeur")}>
            Chauffeured
          </button>
        </div>
      ) : (
        <p className="mt-5 text-sm text-muted">{p.selfPrice !== null ? "Self-drive" : "Chauffeured only"}</p>
      )}

      <div className="mt-5 grid grid-cols-2 gap-3">
        <div>
          <label htmlFor="starts_on" className={lab}>Pick-up</label>
          <input
            id="starts_on" name="starts_on" type="date" required className={box}
            min={today || undefined} value={start}
            onChange={(e) => {
              setStart(e.target.value);
              if (end && e.target.value && end < e.target.value) setEnd("");
            }}
          />
        </div>
        <div>
          <label htmlFor="ends_on" className={lab}>Return</label>
          <input
            id="ends_on" name="ends_on" type="date" required className={box}
            min={start || today || undefined} value={end} onChange={(e) => setEnd(e.target.value)}
          />
        </div>
      </div>

      <p className="mt-2 text-xs text-muted">Minimum {plural(p.minDays, "day")}</p>

      {ranges.length > 0 && (
        <p className="mt-2 text-xs text-muted">
          Fully booked: {ranges.slice(0, 4).map((r) => `${niceDate(r.from)} to ${niceDate(r.to)}`).join(", ")}
          {ranges.length > 4 ? " and more" : ""}
        </p>
      )}

      {problem && (
        <p role="alert" className="mt-3 text-sm underline decoration-accent underline-offset-4">{problem}</p>
      )}

      <div className="mt-4">
        <label htmlFor="guests" className={lab}>Passengers</label>
        <ThemedSelect
          id="guests" name="guests" className={box} defaultValue="1"
          options={Array.from({ length: p.seats }, (_, i): [string, string] => [String(i + 1), String(i + 1)])}
        />
      </div>

      {p.deliveryEnabled && (
        <div className="mt-5 space-y-4">
          <DeliveryBlock
            name="pickup" title="GETTING THE CAR" timeLabel="Delivery time"
            labels={{ none: "Collect the car from us", point: "Deliver to the airport or SGR", custom: "Deliver to a place I choose" }}
            points={p.points} value={pickup} onChange={setPickup} currency={p.currency}
          />
          <DeliveryBlock
            name="dropoff" title="RETURNING THE CAR" timeLabel="Collection time"
            labels={{ none: "Return the car to us", point: "Collect from the airport or SGR", custom: "Collect from a place I choose" }}
            points={p.points} value={dropoff} onChange={setDropoff} currency={p.currency}
          />
          {needsPin && (
            <p className="text-xs text-muted">Please drop a pin on the map for the place you chose.</p>
          )}
        </div>
      )}

      <ContactFields
        pending={pending}
        error={state.error}
        label="Request this car"
        disabled={!!problem || needsPin}
        hasPolicies={p.hasPolicies}
      />

      {total !== null && base !== null && days !== null && (
        <div className="mt-5 space-y-1 border-t border-line pt-4 text-sm">
          <p className="flex justify-between gap-4">
            <span className="text-muted">{plural(days, "day")} at {money(rate, p.currency)}</span>
            <span>{money(base, p.currency)}</span>
          </p>
          {pickupFee > 0 && (
            <p className="flex justify-between gap-4">
              <span className="text-muted">Delivery</span>
              <span>{money(pickupFee, p.currency)}</span>
            </p>
          )}
          {dropoffFee > 0 && (

        ...(() => {
          const known = new Set(locations.map((l) => l.slug));
          const rest = result.items.filter((i) => !i.location || !known.has(i.location.slug));
          return rest.length ? [{ name: "Other", items: rest }] : [];
        })(),
      ];

  function row(item: Item) {
    if (item.kind === "stay") return <StayRow key={item.id} s={item} nights={length} qs={qs} />;
    if (item.kind === "ride") return <RideRow key={item.id} r={item} days={length} from={q.from} to={q.to} />;

            priceMin={result.priceMin}
            priceMax={result.priceMax}
            currency={currency}
          />

          <p className="text-sm text-muted" aria-live="polite">
            {count} {count === 1 ? one : many} {q.hasDates ? "available" : "found"}
            {locName ? ` in ${locName}` : ""}
            {q.hasDates ? ` from ${short(q.from)} to ${short(q.to)}` : ""}
          </p>
          {!q.hasDates && (
            <p className="mt-1 text-xs text-muted">Add your dates above to see only what is free.</p>
          )}

          {count === 0 ? (
            <div className="mt-8 space-y-4 border border-line p-6">
              <p className="font-serif text-xl italic">
                {q.hasDates ? "Nothing is free for those dates." : "Nothing matches these filters."}
              </p>
              <p className="text-sm text-muted">

                help you find something.
              </p>
              <a
                href={WHATSAPP_URL}
                target="_blank"
                rel="noopener noreferrer"
                className="inline-block border border-accent px-5 py-2.5 text-sm tracking-widest transition-colors hover:bg-accent/15"

                <Link href={`/${WORD[type]}`} className="text-sm underline decoration-line underline-offset-4 hover:decoration-accent">
                  Start a new search
                </Link>
              </p>
            </div>
          ) : (
            <div className="mt-8 space-y-12">
              {sections.map((g) => (
                <section key={g.name || "all"} aria-label={g.name || undefined} className="space-y-6">
                  {!q.location && <h2 className="text-2xl">{g.name.toLowerCase()}</h2>}
                  {g.items.map(row)}
                </section>
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}

"use client";

import Link from "next/link";
import { useEffect, useId, useState } from "react";
import { useRouter } from "next/navigation";
import Icon, { type IconName } from "@/components/Icon";
import ThemedSelect from "@/components/ThemedSelect";

type Type = "stay" | "ride" | "tour";
type Loc = { slug: string; name: string };

type Props = {
  locations: Loc[];
  toursReady: boolean;
  initial: { type: Type; location: string; from: string; to: string };
  compact?: boolean;
  /** "choose": the tabs pick a service before searching (home page).
      "navigate": the tabs are links, so the box always matches the page (results pages). */
  mode?: "choose" | "navigate";
};

const SEGMENT: Record<Type, string> = { stay: "stays", ride: "rides", tour: "tours" };
const TABS: { key: Type; label: string; icon: IconName }[] = [
  { key: "stay", label: "Stays", icon: "home" },
  { key: "ride", label: "Rides", icon: "car" },
  { key: "tour", label: "Tours", icon: "compass" },
];
const LABELS: Record<Type, [string, string]> = {
  stay: ["Check-in", "Check-out"],
  ride: ["Pick-up", "Return"],
  tour: ["From", "To (optional)"],
};

const box = "w-full border border-line bg-bg px-3 py-3 text-sm text-fg focus:border-accent focus:outline-none";
const lab = "mb-1.5 block text-xs tracking-widest text-muted";

function addDay(iso: string, n: number) {
  const d = new Date(iso + "T00:00:00Z");
  d.setUTCDate(d.getUTCDate() + n);
  return d.toISOString().slice(0, 10);
}

export default function SearchBox({ locations, toursReady, initial, compact = false, mode = "choose" }: Props) {
  const uid = useId();
  const router = useRouter();
  const [chosen, setChosen] = useState<Type>(initial.type);
  const [location, setLocation] = useState(initial.location);
  const [from, setFrom] = useState(initial.from);
  const [to, setTo] = useState(initial.to);
  const [error, setError] = useState("");
  const [today, setToday] = useState("");

  useEffect(() => setToday(new Date().toLocaleDateString("en-CA")), []);

  // On results pages the box always follows the page, so it can never disagree with it.
  const type: Type = mode === "navigate" ? initial.type : chosen;
  const tabs = TABS.filter((t) => t.key !== "tour" || toursReady);
  const [startLabel, endLabel] = LABELS[type];
  const toMin = from ? (type === "stay" ? addDay(from, 1) : from) : today;

  function hrefFor(t: Type) {
    const p = new URLSearchParams();
    if (location) p.set("location", location);
    if (from) {
      p.set("from", from);
      const keepTo = to && (t === "stay" ? to > from : to >= from);
      if (keepTo) p.set("to", to);
    }
    const q = p.toString();
    return `/${SEGMENT[t]}${q ? `?${q}` : ""}`;
  }

  function submit(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    if (to && !from) return setError("Please choose a start date too.");
    if (from && !to && type !== "tour") return setError(`Please choose your ${endLabel.toLowerCase()} date too.`);
    if (from && to && (type === "stay" ? to <= from : to < from)) {
      return setError(type === "stay" ? "Check-out must be after check-in." : "The end date cannot be before the start date.");
    }
    router.push(hrefFor(type));
  }

  const tabClass = (active: boolean) =>
    `-mb-px inline-flex items-center gap-2 border-b pb-3 text-sm tracking-widest transition-colors ${
      active ? "border-accent text-fg" : "border-transparent text-muted hover:text-fg"
    }`;

  return (
    <form
      onSubmit={submit}
      aria-label="Search"
      className={compact ? "border border-line p-4 sm:p-5" : "border border-line bg-bg p-4 sm:p-5"}
    >
      {mode === "navigate" ? (
        <nav aria-label="What are you looking for" className="mb-5 flex flex-wrap gap-x-7 gap-y-2 border-b border-line">
          {tabs.map((t) => (
            <Link
              key={t.key}
              href={hrefFor(t.key)}
              aria-current={type === t.key ? "page" : undefined}
              className={tabClass(type === t.key)}
            >
              <Icon name={t.icon} size={17} />
              {t.label}
            </Link>
          ))}
        </nav>
      ) : (
        <div role="tablist" aria-label="What are you looking for" className="mb-5 flex flex-wrap gap-x-7 gap-y-2 border-b border-line">
          {tabs.map((t) => (
            <button
              key={t.key}
              type="button"
              role="tab"
              aria-selected={type === t.key}
              onClick={() => setChosen(t.key)}
              className={tabClass(type === t.key)}
            >
              <Icon name={t.icon} size={17} />
              {t.label}
            </button>
          ))}
        </div>
      )}

      <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-[1.3fr_1fr_1fr_auto] lg:items-end">
        <div className="sm:col-span-2 lg:col-span-1">
          <label htmlFor={`${uid}-loc`} className={lab}>Location</label>
          <ThemedSelect
            id={`${uid}-loc`}
            name="location"
            defaultValue={location}
            onChange={setLocation}
            className={box}
            options={[["", "All locations"], ...locations.map((l): [string, string] => [l.slug, l.name])]}
          />
        </div>
        <div>
          <label htmlFor={`${uid}-from`} className={lab}>{startLabel}</label>
          <input
            id={`${uid}-from`} type="date" className={box} min={today || undefined} value={from}
            onChange={(e) => {
              setFrom(e.target.value);
              if (to && e.target.value && (type === "stay" ? to <= e.target.value : to < e.target.value)) setTo("");
            }}
          />
        </div>
        <div>
          <label htmlFor={`${uid}-to`} className={lab}>{endLabel}</label>
          <input id={`${uid}-to`} type="date" className={box} min={toMin || undefined} value={to} onChange={(e) => setTo(e.target.value)} />
        </div>
        <button
          type="submit"
          className="inline-flex items-center justify-center gap-2 bg-accent px-7 py-3 text-sm tracking-widest text-on-accent transition-opacity hover:opacity-90 sm:col-span-2 lg:col-span-1"
        >
          <Icon name="search" size={16} />
          Search
        </button>
      </div>

      {error && (
        <p role="alert" className="mt-4 text-sm underline decoration-accent underline-offset-4">{error}</p>
      )}
    </form>
  );
}

"use client";

import { useEffect, useState } from "react";
import { usePathname, useRouter, useSearchParams } from "next/navigation";
import Icon from "@/components/Icon";
import { money } from "@/lib/format";

type Type = "stay" | "ride" | "tour";
type Props = { type: Type; categories: string[]; priceMin: number; priceMax: number; currency: string };

const SEATS: [string, string][] = [["", "Any"], ["2", "2+"], ["5", "5+"], ["7", "7+"]];
const BEDS: [string, string][] = [["", "Any"], ["1", "1+"], ["2", "2+"], ["3", "3+"]];
const GUESTS: [string, string][] = [["", "Any"], ["2", "2+"], ["4", "4+"], ["6", "6+"]];
const TOUR_TYPES: [string, string][] = [["", "Any"], ["fixed", "Fixed tour"], ["guide", "Guide by the day"]];

function Seg({
  label, options, value, onPick,
}: { label: string; options: [string, string][]; value: string; onPick: (v: string) => void }) {
  return (
    <fieldset className="space-y-3">
      <legend className="mb-3 text-xs tracking-widest text-muted">{label}</legend>
      <div className="flex flex-wrap gap-2">
        {options.map(([v, l]) => (
          <button
            key={v || "any"} type="button" aria-pressed={value === v} onClick={() => onPick(v)}
            className={`border px-3 py-2 text-xs transition-colors ${
              value === v ? "border-accent bg-accent/15 text-fg" : "border-line text-muted hover:border-accent hover:text-fg"
            }`}
          >
            {l}
          </button>
        ))}
      </div>
    </fieldset>
  );
}

export default function SearchFilters({ type, categories, priceMin, priceMax, currency }: Props) {
  const router = useRouter();
  const path = usePathname();
  const sp = useSearchParams();
  const [open, setOpen] = useState(false);
  const fromUrl = Number(sp.get("max")) || priceMax;
  const [price, setPrice] = useState(fromUrl);
  useEffect(() => setPrice(Number(sp.get("max")) || priceMax), [sp, priceMax]);

  const cats = (sp.get("cat") ?? "").split(",").filter(Boolean);

  function update(changes: Record<string, string | null>) {
    const p = new URLSearchParams(sp.toString());
    for (const [k, v] of Object.entries(changes)) {
      if (v === null || v === "") p.delete(k);
      else p.set(k, v);
    }
    const q = p.toString();
    router.replace(q ? `${path}?${q}` : path, { scroll: false });
  }

  const commitPrice = () => update({ max: price >= priceMax ? null : String(price) });
  const toggleCat = (c: string) => {
    const next = cats.includes(c) ? cats.filter((x) => x !== c) : [...cats, c];
    update({ cat: next.length ? next.join(",") : null });
  };

  return (
    <aside aria-label="Filters" className="lg:sticky lg:top-24 lg:self-start">
      <button
        type="button"
        aria-expanded={open}
        onClick={() => setOpen((o) => !o)}
        className="inline-flex items-center gap-2 border border-line px-5 py-3 text-sm tracking-widest lg:hidden"
      >
        <Icon name={open ? "close" : "filter"} size={16} />
        {open ? "Hide filters" : "Filters"}
      </button>

      <div className={`${open ? "mt-4 block" : "hidden"} space-y-8 border border-line p-5 lg:mt-0 lg:block`}>
        <div className="flex items-center justify-between">
          <h2 className="text-lg">filters</h2>
          <button
            type="button"
            onClick={() => update({ max: null, cat: null, seats: null, beds: null, guests: null, ttype: null })}
            className="text-xs tracking-widest text-muted underline decoration-line underline-offset-4 hover:decoration-accent"
          >
            Clear
          </button>
        </div>

        {priceMax > priceMin && (
          <div>
            <label htmlFor="max-price" className="mb-3 block text-xs tracking-widest text-muted">
              {type === "stay" ? "MAX PRICE PER NIGHT" : type === "ride" ? "MAX PRICE PER DAY" : "MAX PRICE"}
            </label>
            <input
              id="max-price" type="range" min={priceMin} max={priceMax} step={100} value={price}
              onChange={(e) => setPrice(Number(e.target.value))}
              onPointerUp={commitPrice} onKeyUp={commitPrice} onBlur={commitPrice}
              className="w-full accent-[var(--accent)]"
            />
            <p className="mt-2 flex justify-between text-xs text-muted">
              <span>{money(priceMin, currency)}</span>
              <span className="text-fg">Up to {money(price, currency)}</span>
            </p>
          </div>
        )}

        {type === "ride" && categories.length > 0 && (
          <fieldset>
            <legend className="mb-3 text-xs tracking-widest text-muted">VEHICLE TYPE</legend>
            <div className="space-y-2.5">
              {categories.map((c) => (
                <label key={c} className="flex items-center gap-3 text-sm">
                  <input type="checkbox" checked={cats.includes(c)} onChange={() => toggleCat(c)} className="h-4 w-4 accent-[var(--accent)]" />
                  {c}
                </label>
              ))}
            </div>
          </fieldset>
        )}

        {type === "ride" && <Seg label="SEATS" options={SEATS} value={sp.get("seats") ?? ""} onPick={(v) => update({ seats: v })} />}
        {type === "stay" && <Seg label="BEDROOMS" options={BEDS} value={sp.get("beds") ?? ""} onPick={(v) => update({ beds: v })} />}
        {type === "stay" && <Seg label="GUESTS" options={GUESTS} value={sp.get("guests") ?? ""} onPick={(v) => update({ guests: v })} />}
        {type === "tour" && <Seg label="TOUR TYPE" options={TOUR_TYPES} value={sp.get("ttype") ?? ""} onPick={(v) => update({ ttype: v })} />}
      </div>
    </aside>
  );
}
